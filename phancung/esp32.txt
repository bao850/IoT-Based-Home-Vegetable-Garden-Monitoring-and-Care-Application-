        while(1) delay(1000);
    }
    // ===== START STREAM =====
//startCameraStream();


    // ===== FIREBASE =====
    config.host = "smartgarden-85ec4-default-rtdb.asia-southeast1.firebasedatabase.app";
    config.signer.tokens.legacy_token = "nJHUyfkUaKHdgZNZONKRs1RDgCxKrDnNMDcyYAbq";
    initFirebase(&config, &auth);


    // ===== STM32 UART INIT =====
    initSTM32(RX_PIN, TX_PIN);
    delay(500);          // ⭐ BẮT BUỘC
    loadThresholdFromFirebase();
delay(200);
sendThresholdToSTM32();
